## Prim

### 最小生成树

```cpp
ll prim(int n, vector<vector<PII>>& graph) {
    vector<bool> visited(n + 1, false);
    vector<int> minDist(n + 1, INT_MAX);
    priority_queue<PII, vector<PII>, greater<PII>> pq;
    ll totalWeight = 0;
    int vertexCount = 0;
    minDist[1] = 0;
    pq.push({0, 1});
    while (!pq.empty() && vertexCount < n) {
        auto [w, u] = pq.top();
        pq.pop();
        if (visited[u]) continue;
        visited[u] = true;
        totalWeight += w;
        vertexCount++;
        for (const auto& [v, weight] : graph[u]) {
            if (!visited[v] && weight < minDist[v]) {
                minDist[v] = weight;
                pq.push({weight, v});
            }
        }
    }
    return (vertexCount == n) ? totalWeight : -1;
}
```

### 状压DP (三人分组)

```cpp
int n;
cin >> n;
int m = 3 * n;
vector<vector<long long>> a(m, vector<long long>(m));
for (int i = 0; i < m; i++) {
    for (int j = 0; j < m; j++) {
        cin >> a[i][j];
    }
}
int totalStates = 1 << m;
vector<long long> dp(totalStates, -1e18);
dp[0] = 0;
for (int mask = 0; mask < totalStates; mask++) {
    if (dp[mask] == -1e18) continue;
    int cnt = __builtin_popcount(mask);
    if (cnt % 3 != 0) continue;
    if (cnt == m) continue;
    int first = -1;
    for (int i = 0; i < m; i++) {
        if (!(mask & (1 << i))) {
            first = i;
            break;
        }
    }
    for (int j = first + 1; j < m; j++) {
        if (mask & (1 << j)) continue;
        for (int k = j + 1; k < m; k++) {
            if (mask & (1 << k)) continue;
            int newMask = mask | (1 << first) | (1 << j) | (1 << k);
            long long addScore = a[first][j] + a[first][k] + a[j][k];
            dp[newMask] = max(dp[newMask], dp[mask] + addScore);
        }
    }
}
cout << dp[totalStates - 1] << "\n";
```

## Dijkstra

### 不记录路径 (邻接矩阵)

```cpp
void dijkstra_no_path(const vector<vector<int>>& graph, int start, vector<int>& dist) {
    int N = graph.size();
    dist.assign(N, INF);
    vector<bool> visited(N, false);
    dist[start] = 0;
    for (int i = 0; i < N; ++i) {
        int u = -1, min_dist = INF;
        for (int j = 0; j < N; ++j) {
            if (!visited[j] && dist[j] < min_dist) {
                min_dist = dist[j];
                u = j;
            }
        }
        if (u == -1) break;
        visited[u] = true;
        for (int v = 0; v < N; ++v) {
            if (!visited[v] && graph[u][v] != INF && dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];
            }
        }
    }
}
```

```cpp
void floyd_with_path(ll n, vector<vector<ll>>& dist, vector<vector<ll>>& next) {
    for (ll i = 1; i <= n; ++i) {
        for (ll j = 1; j <= n; ++j) {
            if (i != j && dist[i][j] != INF) {
                next[i][j] = j;
            } else {
                next[i][j] = -1;
            }
        }
    }
    for (ll k = 1; k <= n; ++k) {
        for (ll i = 1; i <= n; ++i) {
            for (ll j = 1; j <= n; ++j) {
                if (dist[i][k] != INF && dist[k][j] != INF) {
                    if (dist[i][j] > dist[i][k] + dist[k][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                        next[i][j] = next[i][k];
                    }
                }
            }
        }
    }
}

vector<ll> get_path(ll u, ll v, const vector<vector<ll>>& next) {
    vector<ll> path;
    if (next[u][v] == -1) return path;
    path.push_back(u);
    while (u != v) {
        u = next[u][v];
        path.push_back(u);
    }
    return path;
```

## BFS

### 基本BFS (迷宫最短路)

```cpp
ll n, m, t;
ll sx, sy, fx, fy;
ll dx[4] = {0, 0, 1, -1};
ll dy[4] = {1, -1, 0, 0};
bool vec[1001][1001] = {0};
bool vis[1001][1001] = {0};

struct place {
    ll x, y, step;
};

bool check(ll x, ll y) {
    if (x >= 1 && x <= n && y >= 1 && y <= m && vec[x][y] != 1 && vis[x][y] != 1) {
        return 1;
    }
    return 0;
}

int bfs(ll a, ll b) {
    queue<place> st;
    st.push({a, b, 0});
    vis[a][b] = 1;
    while (!st.empty()) {
        place p = st.front();
        st.pop();
        ll px = p.x;
        ll py = p.y;
        for (ll i = 0; i < 4; i++) {
            ll nx = px + dx[i];
            ll ny = py + dy[i];
            if (check(nx, ny)) {
                if (nx == fx && ny == fy) {
                    return p.step + 1;
                } else {
                    st.push({nx, ny, p.step + 1});
                    vis[nx][ny] = 1;
                }
            }
        }
    }
    return -1;
}
```

```cpp
ll fast_pow(ll a, ll b, ll mod) {
    ll res = 1;
    while (b > 0) {
        if (b & 1) res = (res * a) % mod;
        a = (a * a) % mod;
        b >>= 1;
    }

}
```

```cpp
            if (vis[dq[rig]] == 0) ans++;
            vis[dq[rig]]++;
            rig++;
        }
```

```cpp
int main() {
```

```cpp
            dq_min.pop_back();
        }

            dq_max.pop_back();

            dq_max.pop_front();
        }

        if (i >= k - 1) {

            gre.push_back(vec[dq_max.front()]);
        }
    }
```

## 导数零点 + 二分找函数极值点

### 法一

```cpp
const double EPS = 1e-5;

double fast_pow(double a, ll b) {
    double ans = 1.0;
    while (b) {
        if (b & 1) ans *= a;
        a *= a;
        b >>= 1;
    }
    return ans;
}

double cal(const vector<double>& coef, double x, ll n) {
    double result = 0.0;
    for (ll i = 1; i <= n; i++) {
        result += coef[i] * i * fast_pow(x, i - 1);
    }
    return result;
}

int main() {
    cout << fixed << setprecision(5);
    ll n;
    double l, r;
    cin >> n >> l >> r;
    vector<double> coef(n + 1, 0);
    for (ll i = n; i >= 0; i--) {
        cin >> coef[i];
    }
    double L = l, R = r;
    while (R - L > EPS) {
        double mid = (L + R) / 2;
        double der = cal(coef, mid, n);
        if (der > 0) {
            L = mid;
        } else {
            R = mid;
        }
    }
    cout << (L + R) / 2 << endl;
    return 0;
}
```

### 法二

```cpp
const double EPS = 1e-8;
int n;
double l, r;
vector<double> coef;

double f(double x) {
    double res = 0;
    for (int i = 0; i <= n; i++) {
        res = res * x + coef[i];
    }
    return res;
}

double df(double x) {
    double res = 0;
    for (int i = 0; i < n; i++) {
        res = res * x + coef[i] * (n - i);
    }
    return res;
}

        if (df(mid) > 0) {
            L = mid;
        } else {
            R = mid;
        }
    }

}
```
